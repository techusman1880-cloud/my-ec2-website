class TrafficSignalAgent: def __init__(self, name): self.name = name self.signal = "Red" def act(self, traffic): if traffic > 10: self.signal = "Green" else: self.signal = "Red" return f"{self.name}: {self.signal}" class TrafficCoordinator: def __init__(self): self.signals = [ TrafficSignalAgent("Signal A"), TrafficSignalAgent("Signal B") ] def act(self, traffic_data): for signal, traffic in zip(self.signals, traffic_data): print(signal.act(traffic)) coordinator = TrafficCoordinator() coordinator.act([15, 5])
